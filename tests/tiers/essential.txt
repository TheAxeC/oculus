# The tests npm run test:quick always runs, whatever the step changed: those of the behaviour whose
# failure costs most (a write that runs without the page's token, a path outside the served roots,
# a write that changes more than it names, an action that loses work, hub text reaching HTML
# unsanitised, a path reaching a shell), each named by an entry of agents/guarded.md. The list is
# kept short so the quick list holds 30 s; every other test runs in npm run test:landing.
#
# One path per line, relative to oculus's folder, with a note after the # on what it holds. Each
# path must exist and must not be tagged slow: test:quick reports a path that names no file as a
# failed command. tests/checks is left to npm run test:checks, and the slow files to npm run
# test:slow.

tests/routes/edit/write.test.ts            # write route: token, unreadable record file
tests/server/files/ops.test.ts             # file operations: token, paths outside the roots, trash
tests/server/files/ops-paste.test.ts       # paste and move: replace trashes first
tests/server/files/fs.test.ts              # directory listing: paths that leave a root
tests/routes/git/git.test.ts               # git route: token, confirm for lossy actions
tests/server/edit/config-value.test.ts     # config writer: local.yml refused, stale base
tests/server/edit/writes.test.ts           # each write changes only the file and key it names
tests/ui/html.dom.test.ts                  # the sanitiser every hub text passes through
tests/routes/files/open-program.test.ts    # open: the path is an argument, never a shell string
tests/shell/panes/strip/tab-menu.test.ts   # closing asks before it drops unsaved text
