The measuring script (save it outside the worktree, run from `tools/oculus`):

```python
import re, os, collections
files = [os.path.normpath(os.path.join(d, f)) for d, _, fs in os.walk('src') for f in fs if f.endswith(('.ts', '.svelte'))]
known = set(files)
def res(frm, spec):
    if spec.startswith('$lib/'): base = os.path.join('src/lib', spec[5:])
    elif spec.startswith('.'): base = os.path.normpath(os.path.join(os.path.dirname(frm), spec))
    else: return None
    for c in (base, base + '.ts', base + '.svelte', base + '/index.ts'):
        if os.path.normpath(c) in known: return os.path.normpath(c)
imp = re.compile(r'''(?:import|export)\s+(?:type\s+)?(?:[^'"]*?\s+from\s+)?['"]([^'"]+)['"]''')
g = collections.defaultdict(set)
for f in files:
    for m in imp.finditer(open(f).read()):
        if re.match(r'(import|export)\s+type', m.group(0)): continue
        r = res(f, m.group(1))
        if r: g[f].add(r)
def scc(graph, nodes):
    idx, low, st, on, out, i = {}, {}, [], set(), [], [0]
    def visit(v):
        idx[v] = low[v] = i[0]; i[0] += 1; st.append(v); on.add(v)
        for w in graph.get(v, ()):
            if w not in idx: visit(w); low[v] = min(low[v], low[w])
            elif w in on: low[v] = min(low[v], idx[w])
        if low[v] == idx[v]:
            c = []
            while True:
                w = st.pop(); on.discard(w); c.append(w)
                if w == v: break
            if len(c) > 1: out.append(sorted(c))
    for v in nodes:
        if v not in idx: visit(v)
    return out
def folder(p):
    parts = p.replace('src/lib/', '').replace('src/', '').split('/')
    return '/'.join(parts[:2]) if len(parts) > 2 else parts[0]
fg = collections.defaultdict(set)
for a, bs in g.items():
    for b in bs:
        if folder(a) != folder(b): fg[folder(a)].add(folder(b))
print('modules', len(files), 'edges', sum(len(x) for x in g.values()))
print('module cycles', len(scc(g, files)), scc(g, files))
print('folder rings', scc(fg, list(fg)))
e = [(a, b) for a, bs in g.items() for b in bs if a.startswith('src/lib/views') and b.startswith('src/lib/state')]
print('views->state', len(e), 'files', len({a for a, _ in e}), 'targets', len({b for _, b in e}))
```
